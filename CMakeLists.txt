cmake_minimum_required(VERSION 3.16)
project(c_cpp_portfolio LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(BUILD_MPI "Build the MPI example" ON)
option(BUILD_OPENGL "Build the optional OpenGL example" OFF)
if(MSVC)
  add_compile_options(/W4 /WX)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  if(ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      add_compile_options(-fno-pie)
      add_link_options(-no-pie)
    endif()
  endif()
endif()

add_executable(bankers c/bankers/bankers_algorithm.c)
if(UNIX)
  find_package(Threads REQUIRED)
  add_executable(process_info c/posix/process_info.c)
  foreach(mode IN ITEMS with without)
    add_executable(threads_${mode}_semaphore c/posix/thread_output_${mode}_semaphore.c)
    target_link_libraries(threads_${mode}_semaphore PRIVATE Threads::Threads)
  endforeach()
endif()
if(BUILD_MPI)
  find_package(MPI REQUIRED COMPONENTS C)
  add_executable(mpi_minimum c/mpi-minimum/mpi-minimum.c)
  target_link_libraries(mpi_minimum PRIVATE MPI::MPI_C)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  file(GLOB UNIT_TESTS CONFIGURE_DEPENDS tests/test_*.cpp tests/test_*.c)
  foreach(source IN LISTS UNIT_TESTS)
    get_filename_component(name ${source} NAME_WE)
    add_executable(${name} ${source})
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
  add_test(NAME cli_regressions COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE_DIR:bankers>)
  if(BUILD_MPI)
    add_test(NAME mpi_regressions COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_SOURCE_DIR}/tests/test_mpi.py $<TARGET_FILE:mpi_minimum> ${MPIEXEC_EXECUTABLE})
    set_tests_properties(mpi_regressions PROPERTIES TIMEOUT 120 ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  endif()
  if(BUILD_OPENGL)
    find_program(XVFB_RUN xvfb-run REQUIRED)
    add_test(NAME opengl_pixels COMMAND ${XVFB_RUN} -a $<TARGET_FILE:opengl_shapes> --smoke-test)
    set_tests_properties(opengl_pixels PROPERTIES TIMEOUT 30 ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;ASAN_OPTIONS=detect_leaks=0")
  endif()
endif()
